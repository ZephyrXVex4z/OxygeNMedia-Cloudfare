// CLOUDFLARE WORKER — sube imágenes a R2 en nombre de un usuario autenticado
// (Firebase Auth), sin exponer ninguna credencial de R2 en el navegador.
// El binding nativo de R2 hace innecesario el Access Key/Secret de R2 — el
// Worker habla con el bucket directamente.
//
// AHORA SIRVE DOS APPS EN EL MISMO WORKER (mismo bucket, carpetas separadas):
//   - OxygeNMedia  (proyecto Firebase: workwebschool-5646f)
//   - NovaStore    (proyecto Firebase: nova-store-ink)
// El comportamiento de OxygeNMedia queda exactamente igual que antes; solo se
// añadió una capa de configuración por app para no duplicar el Worker entero.
//
// SIN DEPENDENCIAS NPM: la verificación del ID Token de Firebase se hace con
// la Web Crypto API nativa de Cloudflare Workers (crypto.subtle).
//
// FLUJO:
// 1. El navegador manda la imagen aquí junto con su ID Token de Firebase y
//    un header X-App indicando de qué app viene ("oxygenmedia" | "novastore").
// 2. Este Worker verifica el ID Token contra el JWKS público de Google, usando
//    el FIREBASE_PROJECT_ID que corresponda a esa app.
// 3. Verifica que el archivo sea del tipo/tamaño permitido para esa app.
// 4. Lo sube a R2 bajo una ruta organizada por app y tipo:
//      OxygeNMedia -> perfiles/{uid}/{timestamp}.webp, posts/{uid}/{timestamp}.webp, ...
//      NovaStore   -> productos-novastore/{uid}/{timestamp}.{ext}
// 5. Devuelve la URL pública final.
//
// CÓMO DESPLEGARLO (editor simple del dashboard, sin terminal):
// 1. En tu Worker, vista "Edit code" (</>).
// 2. Reemplaza TODO el contenido por este archivo. Guarda/Deploy.
// 3. Dashboard del Worker → Settings → Bindings → confirma que sigue el mismo
//    binding R2 que ya tenías:
//      - Variable name: BUCKET_IMAGENES

// 4. Dashboard del Worker → Settings → Variables and Secrets → agrega (además
//    de FIREBASE_PROJECT_ID que ya tenías, que ahora es el de OxygeNMedia):
//      - FIREBASE_PROJECT_ID_NOVASTORE (tipo normal) = nova-store-ink
// 5. URL_PUBLICA_BASE sigue igual (mismo bucket para ambas apps).
// 6. Deploy. En NovaStore, admin.html debe mandar el header X-App: novastore

//
// NOTA SOBRE CORS: cada app tiene una lista "origenesPermitidos" (no un solo
// string). Mientras pruebas NovaStore desde oxygenmedia.online, ya está
// incluido ahí junto con novastore.ink. Cuando dejes de necesitar probarlo
// desde oxygenmedia.online, puedes quitarlo de esa lista.

const URL_PUBLICA_BASE = "https://pub-71d10c040e384f0ebac013dfb1faa046.r2.dev";

// ---------- Configuración por app ----------
// Cada app define: su project id (de dónde sale el FIREBASE_PROJECT_ID a usar),
// el origen permitido (CORS), el tamaño máximo de archivo, y los tipos/rutas
// válidos que puede subir.
const CONFIG_APPS = {
  oxygenmedia: {
    origenesPermitidos: ["https://oxygenmedia.online"],
    maxBytes: 1 * 1024 * 1024, // 1 MB, ya en WebP
    tiposPermitidos: new Set(["perfiles", "posts", "recursos", "banners"]),
    contentTypesPermitidos: new Set(["image/webp"]),
    // ruta: {tipo}/{uid}/{timestamp}.webp
    construirRuta: (tipo, uid) => `${tipo}/${uid}/${Date.now()}.webp`,
  },
  novastore: {
    origenesPermitidos: ["https://novastore.ink", "https://oxygenmedia.online"],
    maxBytes: 5 * 1024 * 1024, // 5 MB — fotos de producto, no tan comprimidas como perfiles
    tiposPermitidos: new Set(["productos"]),
    contentTypesPermitidos: new Set(["image/webp", "image/jpeg", "image/png"]),
    // ruta: productos-novastore/{uid}/{timestamp}.{ext}
    construirRuta: (tipo, uid, contentType) => {
      const ext = contentType === "image/png" ? "png" : contentType === "image/jpeg" ? "jpg" : "webp";
      return `productos-novastore/${uid}/${Date.now()}.${ext}`;
    },
  },
};

// Cooldown básico anti-spam por usuario (en memoria del Worker).
const ultimaSubidaPorUid = new Map();
const COOLDOWN_MS = 3000;

// ============ VERIFICACIÓN DE ID TOKEN DE FIREBASE (Web Crypto, sin libs) ============

let cacheJwks = null;
let cacheJwksExpira = 0;

async function obtenerJwksFirebase() {
  if (cacheJwks && Date.now() < cacheJwksExpira) return cacheJwks;
  const resp = await fetch("https://www.googleapis.com/service_accounts/v1/jwk/securetoken@system.gserviceaccount.com");
  if (!resp.ok) throw new Error("No se pudieron obtener las claves públicas de Google.");
  const data = await resp.json();
  cacheJwks = data.keys;
  cacheJwksExpira = Date.now() + 60 * 60 * 1000;
  return cacheJwks;
}

function base64UrlADecodificado(str) {
  str = str.replace(/-/g, "+").replace(/_/g, "/");
  while (str.length % 4) str += "=";
  const binario = atob(str);
  return Uint8Array.from(binario, c => c.charCodeAt(0));
}

async function verificarIdTokenFirebase(idToken, projectId) {
  const partes = idToken.split(".");
  if (partes.length !== 3) throw new Error("Formato de token inválido.");
  const [headerB64, payloadB64, firmaB64] = partes;

  const header = JSON.parse(new TextDecoder().decode(base64UrlADecodificado(headerB64)));
  const payload = JSON.parse(new TextDecoder().decode(base64UrlADecodificado(payloadB64)));

  const ahoraSegundos = Math.floor(Date.now() / 1000);
  if (payload.exp < ahoraSegundos) throw new Error("El token expiró.");
  if (payload.aud !== projectId) throw new Error("El token no pertenece a este proyecto.");
  if (payload.iss !== `https://securetoken.google.com/${projectId}`) throw new Error("Emisor de token inválido.");
  if (!payload.sub) throw new Error("Token sin usuario.");

  const jwks = await obtenerJwksFirebase();
  const jwk = jwks.find(k => k.kid === header.kid);
  if (!jwk) throw new Error("No se encontró la clave de firma correspondiente.");

  const clavePublica = await crypto.subtle.importKey(
    "jwk",
    jwk,
    { name: "RSASSA-PKCS1-v1_5", hash: "SHA-256" },
    false,
    ["verify"]
  );

  const datosFirmados = new TextEncoder().encode(`${headerB64}.${payloadB64}`);
  const firma = base64UrlADecodificado(firmaB64);

  const valido = await crypto.subtle.verify("RSASSA-PKCS1-v1_5", clavePublica, firma, datosFirmados);
  if (!valido) throw new Error("Firma del token inválida.");

  return payload.sub;
}

// ============ WORKER ============

export default {
  async fetch(request, env) {
    // Determina qué app está llamando. Por compatibilidad, si no manda
    // X-App, se asume "oxygenmedia" (comportamiento previo intacto).
    const appId = (request.headers.get("X-App") || "oxygenmedia").toLowerCase();
    const configApp = CONFIG_APPS[appId];

    if (!configApp) {
      return jsonError("App desconocida.", 400, { "Access-Control-Allow-Origin": "*" });
    }

    const origenPeticion = request.headers.get("Origin") || "";
    const origenAUsar = configApp.origenesPermitidos.includes(origenPeticion)
      ? origenPeticion
      : configApp.origenesPermitidos[0];

    const headersCORS = {
      "Access-Control-Allow-Origin": origenAUsar,
      "Access-Control-Allow-Methods": "POST, OPTIONS",
      "Access-Control-Allow-Headers": "Content-Type, X-Firebase-Token, X-Tipo-Imagen, X-App",
    };

    if (request.method === "OPTIONS") return new Response(null, { headers: headersCORS });
    if (request.method !== "POST") return jsonError("Método no permitido.", 405, headersCORS);

    const idToken = request.headers.get("X-Firebase-Token");
    const tipo = request.headers.get("X-Tipo-Imagen");

    if (!idToken) return jsonError("Falta el token de sesión.", 401, headersCORS);
    if (!tipo || !configApp.tiposPermitidos.has(tipo)) {
      return jsonError("Tipo de imagen inválido.", 400, headersCORS);
    }

    // Elige el project id de Firebase según la app.
    const projectId = appId === "novastore" ? env.FIREBASE_PROJECT_ID_NOVASTORE : env.FIREBASE_PROJECT_ID;
    if (!projectId) {
      return jsonError("El Worker no tiene configurado el proyecto de Firebase para esta app.", 500, headersCORS);
    }

    let uid;
    try {
      uid = await verificarIdTokenFirebase(idToken, projectId);
    } catch (err) {
      return jsonError("Sesión inválida o expirada: " + err.message, 401, headersCORS);
    }

    const claveCooldown = `${appId}:${uid}`;
    const ahora = Date.now();
    const ultima = ultimaSubidaPorUid.get(claveCooldown) || 0;
    if (ahora - ultima < COOLDOWN_MS) {
      return jsonError("Espera un momento antes de subir otra imagen.", 429, headersCORS);
    }

    const bytes = await request.arrayBuffer();
    if (bytes.byteLength === 0) return jsonError("No se recibió ningún archivo.", 400, headersCORS);
    if (bytes.byteLength > configApp.maxBytes) {
      const maxMB = (configApp.maxBytes / 1024 / 1024).toFixed(1);
      return jsonError(`La imagen pesa demasiado (máximo ${maxMB} MB, esta pesa ${(bytes.byteLength / 1024 / 1024).toFixed(2)} MB).`, 400, headersCORS);
    }

    const contentType = request.headers.get("Content-Type") || "";
    if (!configApp.contentTypesPermitidos.has(contentType)) {
      return jsonError("Formato de imagen no permitido para esta app.", 400, headersCORS);
    }

    const nombreArchivo = configApp.construirRuta(tipo, uid, contentType);
    try {
      await env.BUCKET_IMAGENES.put(nombreArchivo, bytes, {
        httpMetadata: { contentType },
      });
    } catch (err) {
      return jsonError("No se pudo guardar la imagen: " + err.message, 502, headersCORS);
    }

    ultimaSubidaPorUid.set(claveCooldown, ahora);

    const urlPublica = `${URL_PUBLICA_BASE}/${nombreArchivo}`;
    return new Response(JSON.stringify({ url: urlPublica }), {
      headers: { ...headersCORS, "Content-Type": "application/json" },
    });
  },
};

function jsonError(mensaje, status, headersCORS) {
  return new Response(JSON.stringify({ error: mensaje }), {
    status,
    headers: { ...headersCORS, "Content-Type": "application/json" },
  });
}
